# BlackBox MCP Server Dependencies
# Kept in a SEPARATE venv (MCP/venv/) from Orchestrator/venv/ because mcp's
# transitive starlette>=0.49 conflicts with fastapi 0.118's starlette<0.49
# upper bound. Two venvs is the cost of isolation.
#
# This venv serves the Funnel-exposed (public internet) HTTP transport, so every
# package that parses attacker-controlled bytes carries a SECURITY FLOOR (the
# first release with the fix) and a cap below its next major. Floors matter on
# EXISTING boxes: the updater's mcp_pip phase and install.sh run a plain
# `pip install -r` (no --upgrade), so only a raised floor moves an installed
# version. Bump a floor when an advisory lands; never drop a cap without
# re-running MCP/test_*.py against the new major.

# mcp 2.x removed the low-level @server.list_tools()/@server.call_tool()
# decorators this server registers its handlers with, so a fresh install
# without the cap crashes at import. 1.27.2 fixes CVE-2026-52869 (sessions not
# bound to the authenticated principal) and CVE-2026-52870; 1.30.0 fixes
# CVE-2026-59951 (unbounded Streamable HTTP session accumulation).
mcp>=1.30.0,<2
httpx>=0.25.0
requests>=2.33.0          # Orchestrator/web_tools.py (web_fetch); CVE-2024-47081, CVE-2026-25645
urllib3>=2.8.0,<3         # requests' transport for web_fetch: decompression-bomb + redirect-header CVEs
idna>=3.15                # web_fetch hostname encoding; CVE-2026-45409
beautifulsoup4>=4.12.0    # Orchestrator/web_tools.py (HTML parsing for web_fetch)
lxml>=6.1.3               # the parser web_fetch asks bs4 for; CVE-2026-41066 fixed in 6.1.0

# M2 (Streamable HTTP transport): the SDK's StreamableHTTPSessionManager runs in
# a Starlette ASGI app served by uvicorn. Both are transitive deps of `mcp`;
# they are listed here to make the HTTP transport's requirement EXPLICIT and to
# carry the security floors. fastapi is deliberately NOT added (that is the
# starlette-version conflict the lean venv exists to avoid).
starlette>=1.3.1,<2       # ASGI app + the public /authorize and /token form parsing; CVE-2026-48710, CVE-2026-54283
python-multipart>=0.0.32  # starlette's form parser for /authorize + /token; CVE-2026-40347/-42561/-53537..-53540
uvicorn>=0.38.0,<1        # ASGI server for the HTTP transport (the version the shipped boxes run)
h11>=0.16.0               # uvicorn's HTTP/1.1 parser on the public port; CVE-2025-43859 (chunked smuggling)

# The SDK's own transitive dependencies (X01-deps-python-04). mcp only asks for
# pyjwt[crypto]>=2.10.1 and anyio>=4.5, which an existing venv already
# satisfies, so without these floors a box upgraded in place keeps the
# vulnerable builds. cryptography is left uncapped on purpose: it ships a new
# major several times a year and security fixes land only on the newest one.
PyJWT>=2.15.0,<3          # mcp's JWT layer; CVE-2026-32597, CVE-2026-48522..-48526, CVE-2026-101917/-101918, CVE-2026-102267..-102275
cryptography>=50.0.0      # via pyjwt[crypto]; CVE-2026-26007, -34073, -39892, -69247..-69249 + the bundled-OpenSSL advisory (48.0.1)
anyio>=4.14.2,<5          # the SDK's and starlette's async layer; CVE-2026-63374 (TLSStream IDNA), CVE-2026-64847
